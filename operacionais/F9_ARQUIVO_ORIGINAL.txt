fr_pavio_inf_pct_2 = (fr_range_2 > 0) and (fr_pavio_inf_2 / fr_range_2 * 100) or 0

-- Fractal classico confirmado: a vela [2] e o extremo, com duas velas de cada lado.
fractal_topo_base =
    (high[2] > high[1]) and
    (high[2] > high) and
    (high[2] > high[3]) and
    (high[2] > high[4])

fractal_fundo_base =
    (low[2] < low[1]) and
    (low[2] < low) and
    (low[2] < low[3]) and
    (low[2] < low[4])

-- Em vez de exigir que o pivo seja exatamente a maxima/minima do Donchian,
-- aceita uma pequena margem. Isso reduz os "buracos" que vimos nos fundos/topos bons.
fr_topo_na_borda = (fr_local_high - high[2]) <= fr_margem_borda
fr_fundo_na_borda = (low[2] - fr_local_low) <= fr_margem_borda

-- Prominencia do pivo em relacao aos vizinhos imediatos.
fr_prom_topo = high[2] - max(high[1], high[3])
fr_prom_fundo = min(low[1], low[3]) - low[2]
fr_topo_destacado = fr_prom_topo >= fr_prom_min
fr_fundo_destacado = fr_prom_fundo >= fr_prom_min

-- Houve deslocamento real ate o pivo? Ajuda a nao desenhar F no meio de ruido.
fr_chegada_topo =
    (close[2] > close[4]) or
    ((high[2] - low[4]) >= fr_media_range * 1.10)

fr_chegada_fundo =
    (close[2] < close[4]) or
    ((high[4] - low[2]) >= fr_media_range * 1.10)

-- Confirmacao de rejeicao/afastamento depois do extremo.
fr_rejeicao_topo =
    (fr_pavio_sup_pct_2 >= tonumber(fractal_rejeicao_pct_txt)) or
    (close[1] < close[2]) or
    (close < close[1])

fr_rejeicao_fundo =
    (fr_pavio_inf_pct_2 >= tonumber(fractal_rejeicao_pct_txt)) or
    (close[1] > close[2]) or
    (close > close[1])

-- Lateralizacao apertada: fractal comum vira ruido.
-- Nela, so aceitamos extremos realmente destacados e na borda do range.
fr_range_suficiente =
    (fr_media_range > 0) and
    (fr_amplitude_9 >= fr_media_range * tonumber(fractal_amplitude_mult_txt))

fr_lateral_apertada =
    (alternancias_5 >= 3) and
    (eficiencia_fluxo_6 < tonumber(s8_fluxo_eficiencia_txt)) and
    (not fr_range_suficiente)

fr_topo_valido_em_lateral =
    fr_topo_na_borda and fr_topo_destacado and fr_rejeicao_topo and fr_range_suficiente

fr_fundo_valido_em_lateral =
    fr_fundo_na_borda and fr_fundo_destacado and fr_rejeicao_fundo and fr_range_suficiente

fractal_topo_forte =
    fractal_topo_base and
    fr_topo_na_borda and
    fr_chegada_topo and
    fr_rejeicao_topo and
    (fr_topo_destacado or (fr_pavio_sup_pct_2 >= tonumber(fractal_rejeicao_pct_txt))) and
    ((not fr_lateral_apertada) or fr_topo_valido_em_lateral)

fractal_fundo_forte =
    fractal_fundo_base and
    fr_fundo_na_borda and
    fr_chegada_fundo and
    fr_rejeicao_fundo and
    (fr_fundo_destacado or (fr_pavio_inf_pct_2 >= tonumber(fractal_rejeicao_pct_txt))) and
    ((not fr_lateral_apertada) or fr_fundo_valido_em_lateral)

plot_shape(fractal_topo_forte,
