-- Base preservada: B2 original enviado pelo usuario.
-- H (hb..hy/hbu/hbd/hl) transplantado SEM ALTERACOES.

-- REV: marcador informativo confirmado apos H, nunca um novo setup de entrada.
-- TESTE em demonstracao: nao representa validacao estatistica ou de plataforma.
instrument{name="B2 PA07 H REV TESTE",icon="jpg",overlay=true}
input_group{"Periodo Max/Min",doch_time=input{default="10",type=input.string}}
input_group{"Periodo da Micro Tendencia",emaa_per=input{default="10",type=input.string}}
input_group{"Candles normais",positivo=input{default="#00FF40",type=input.color},negativo=input{default="#FF0000",type=input.color},neutro=input{default="#FFFFFF",type=input.color}}
input_group{"Canal M1",mostrar_canal=input{default=true,type=input.boolean},upline_color=input{default="#FF8C00",type=input.color},lowline_color=input{default="#00FF66",type=input.color}}
input_group{"Sinais",call_color=input{default="#00FF40",type=input.color},put_color=input{default="#FF0000",type=input.color},rev_color=input{default="#00E5FF",type=input.color}}
input_group{"PA07 - lateral ondulada",pa07_on=input{default=true,type=input.boolean}}
input_group{"REV - reversao apos H",rev_on=input{default=true,type=input.boolean}}
input_group{"Fluxo essencial",min_corpo_pct=input{default="20",type=input.string},eficiencia_min_txt=input{default="12",type=input.string},eficiencia_curta_txt=input{default="8",type=input.string},s8_fluxo_eficiencia_txt=input{default="18",type=input.string}}
input_group{"Aviso de Extremo H",extremo_proximidade_txt=input{default="30",type=input.string}}
input_group{"Barras M15",m15_on=input{default=true,type=input.boolean},m15_color=input{default="#FF3EB5",type=input.color},m15_width=input{default=2,type=input.line_width}}

EMAA=ema(close,emaa_per)
upper=highest(high,doch_time)
lower=lowest(low,doch_time)
TA=(close>close[1])and(close>EMAA)and(EMAA>EMAA[1])
TB=(close<close[1])and(close<EMAA)and(EMAA<EMAA[1])
cor_barra=close>open and positivo or(close<open and negativo or neutro)
plot_candle(open,high,low,close,"B2 CANDLES",cor_barra)
if mostrar_canal then
 plot(upper,"Resistencia",upline_color)
 plot(lower,"Suporte",lowline_color)
end
local sec15=security(current_ticker_id,"15m")
if m15_on and sec15 then
 plot(sec15.high[1],"M15 MAX",m15_color,m15_width,0,style.levels,na_mode.continue)
 plot(sec15.low[1],"M15 MIN",m15_color,m15_width,0,style.levels,na_mode.continue)
end

corpo_atual=abs(close-open)
range_atual=high-low
range_1=high[1]-low[1]
range_2=high[2]-low[2]
range_3=high[3]-low[3]
range_4=high[4]-low[4]
range_5=high[5]-low[5]
range_6=high[6]-low[6]
range_7=high[7]-low[7]
range_8=high[8]-low[8]
range_9=high[9]-low[9]
range_10=high[10]-low[10]
range_11=high[11]-low[11]
range_12=high[12]-low[12]
range_13=high[13]-low[13]
range_14=high[14]-low[14]
media_range=(range_atual+range_1+range_2+range_3)/4
media_micro_fechada=(range_1+range_2+range_3+range_4+range_5)/5
media_range_s8=media_micro_fechada
media_range_15=(range_atual+range_1+range_2+range_3+range_4+range_5+range_6+range_7+range_8+range_9+range_10+range_11+range_12+range_13+range_14)/15
pavio_superior_atual=high-max(open,close)
pavio_inferior_atual=min(open,close)-low
pavio_superior_pct=range_atual>0 and pavio_superior_atual/range_atual*100 or 0
pavio_inferior_pct=range_atual>0 and pavio_inferior_atual/range_atual*100 or 0
margem_extremo=media_range*tonumber(extremo_proximidade_txt)/100
perto_resistencia=(upper-high)<=margem_extremo
perto_suporte=(low-lower)<=margem_extremo

-- Flow S8 mantido: alternancias 1x1 de baixa eficiencia.
local t1=close[1]>open[1]and 1 or(close[1]<open[1]and -1 or 0)
local t2=close[2]>open[2]and 1 or(close[2]<open[2]and -1 or 0)
local t3=close[3]>open[3]and 1 or(close[3]<open[3]and -1 or 0)
local t4=close[4]>open[4]and 1 or(close[4]<open[4]and -1 or 0)
local t5=close[5]>open[5]and 1 or(close[5]<open[5]and -1 or 0)
local alternancias_5=(t1~=0 and t2~=0 and t1~=t2 and 1 or 0)+(t2~=0 and t3~=0 and t2~=t3 and 1 or 0)+(t3~=0 and t4~=0 and t3~=t4 and 1 or 0)+(t4~=0 and t5~=0 and t4~=t5 and 1 or 0)
local mov_6=range_atual+range_1+range_2+range_3+range_4+range_5
local eficiencia_fluxo=mov_6>0 and abs(close-close[5])/mov_6*100 or 100
falso_fluxo_s8=alternancias_5>=3 and eficiencia_fluxo<tonumber(s8_fluxo_eficiencia_txt)
local mov_8=range_atual+range_1+range_2+range_3+range_4+range_5+range_6+range_7
local eficiencia_8=mov_8>0 and abs(close-close[7])/mov_8*100 or 100
local mov_4=range_atual+range_1+range_2+range_3
local eficiencia_4=mov_4>0 and abs(close-close[3])/mov_4*100 or 100
local lateral_fraca=eficiencia_8<tonumber(eficiencia_min_txt)or eficiencia_4<tonumber(eficiencia_curta_txt)
local micro_mov=range_1+range_2+range_3+range_4+range_5
local micro_ef=micro_mov>0 and abs(close[1]-close[5])/micro_mov*100 or 100
local caixa_travada=micro_ef<23 and eficiencia_4<18 and alternancias_5>=3

-- PA07: transposto do FS01.5 (regime fechado, sem a vela atual).
local pa_reg,pa_compra,pa_venda=false,false,false
if pa07_on and get_value(close[22],nil)~=nil then
 local sg=security(current_ticker_id,"5m")
 if sg then
  local mo,mc,mh,ml={},{},{},{}
  local s5,ov,ok=0,0,true
  for i=1,5 do
   mo[i]=get_value(sg.open[i],nil);mc[i]=get_value(sg.close[i],nil)
   mh[i]=get_value(sg.high[i],nil);ml[i]=get_value(sg.low[i],nil)
   if mo[i]==nil or mc[i]==nil or mh[i]==nil or ml[i]==nil then ok=false
   else s5=s5+max(0,mh[i]-ml[i]) end
  end
  if ok and s5>0 then
   for i=1,4 do
    ov=ov+max(0,min(mh[i],mh[i+1])-max(ml[i],ml[i+1]))/max(.000000001,min(mh[i]-ml[i],mh[i+1]-ml[i+1]))
   end
   local f=0
   local topo,fundo=high[1],low[1]
   local bw=0
   for i=2,21 do f=f+max(0,high[i]-low[i]) end
   f=f/20
   for i=2,20 do topo=max(topo,high[i]);fundo=min(fundo,low[i]) end
   local largura=topo-fundo
   local s6,s8=0,0
   for i=1,6 do s6=s6+max(0,high[i]-low[i]) end
   for i=1,8 do s8=s8+max(0,high[i]-low[i]) end
   for i=1,19 do
    local a,b=max(0,high[i]-low[i]),max(0,high[i+1]-low[i+1])
    bw=bw+max(0,min(high[i],high[i+1])-max(low[i],low[i+1]))/max(.000000001,min(a,b))
   end
   bw=bw/19
   local prog=close[1]-open[8]
   local alinha=0
   for i=1,8 do
    if(prog>0 and close[i]>open[i])or(prog<0 and close[i]<open[i])then alinha=alinha+1 end
   end
   local esc=s8>0 and abs(prog)/s8>.24 and alinha>=6
   local pernas,anterior,comprimento=0,0,0
   for i=12,1,-1 do
    local direcao=close[i]>open[i]and 1 or(close[i]<open[i]and -1 or 0)
    if direcao~=0 then
     if direcao==anterior then comprimento=comprimento+1
     else
      if comprimento>=2 and comprimento<=5 then pernas=pernas+1 end
      anterior=direcao;comprimento=1
     end
    end
   end
   if comprimento>=2 and comprimento<=5 then pernas=pernas+1 end
   local alta,baixa=high[3],low[3]
   for i=4,22 do alta=max(alta,high[i]);baixa=min(baixa,low[i]) end
   local rompe=(close[1]>alta+f*.15 and close[2]>alta+f*.15)or(close[1]<baixa-f*.15 and close[2]<baixa-f*.15)
   local lat5=abs(mc[1]-mo[5])/s5<.33 and ov/4>=.57
   local lat1=s6>0 and(abs(close[1]-close[6])/s6<.35 or bw>=.45)
   local larga=f>0 and largura>=f*3 and largura<=f*8.5
   pa_reg=lat5 and lat1 and larga and pernas>=2 and not esc and not rompe
   if pa_reg then
    local pos=(close[1]-fundo)/largura
    pa_compra=pos>=.72 and close[1]>close[3]
    pa_venda=pos<=.28 and close[1]<close[3]
   end
  end
 end
end

-- Motor essencial B2: EMA10, forca, fluxo e microcaixote.
-- Removidos: freio S6 pos impulso, transicao S6, S9 lite, F9 extra,
-- reentrada extra, rearme 3 velas/R?, veto de 3 pavios e alerta visual duplicado.
local forca_ok=corpo_atual>media_range_15*tonumber(min_corpo_pct)/100
local vela_qualificada=range_atual>0 and corpo_atual/range_atual>=.45
local sinal_compra=TA and forca_ok and not lateral_fraca and not falso_fluxo_s8 and not caixa_travada and vela_qualificada and pavio_superior_pct<35 and not pa_compra
local sinal_venda=TB and forca_ok and not lateral_fraca and not falso_fluxo_s8 and not caixa_travada and vela_qualificada and pavio_inferior_pct<35 and not pa_venda

-- H original B2 (nao alterar os parametros nem a maquina de estados).
hb=(range_1+range_2+range_3+range_4+range_5)/5
hc=(range_atual>0)and(corpo_atual/range_atual*100)or 0
ht=max(high[1],max(high[2],high[3]))
hf=min(low[1],min(low[2],low[3]))
hpa=(hb>0)and((close[1]-close[6])>=hb*1.50)and(close[1]>close[3])
hpb=(hb>0)and((close[6]-close[1])>=hb*1.50)and(close[1]<close[3])
hes=(high>=ht)and((high-ht)<=hb*0.20)
heb=(low<=hf)and((hf-low)<=hb*0.20)
hfs=(high[1]>=max(high[2],high[3]))and(high<=high[1]+hb*0.05)
hfb=(low[1]<=min(low[2],low[3]))and(low>=low[1]-hb*0.05)
hru=(close>ht+hb*0.05)and(close>open)and(hc>=55)and(pavio_superior_pct<=30)
hrd=(close<hf-hb*0.05)and(close<open)and(hc>=55)and(pavio_inferior_pct<=30)
hps=(close<=close[1]+hb*0.08)or(pavio_superior_pct>=30)or(hc<45)
hpbx=(close>=close[1]-hb*0.08)or(pavio_inferior_pct>=30)or(hc<45)
hrs=hpa and perto_resistencia and(hes or hfs)and hps and(not hru)
hrb=hpb and perto_suporte and(heb or hfb)and hpbx and(not hrd)
local hl=make_series()
local htl=make_series()
local hfl=make_series()
local hage=make_series()
hl:set(nz(hl[1],0))
htl:set(nz(htl[1],ht))
hfl:set(nz(hfl[1],hf))
hage:set(nz(hage[1],0))
h0=nz(hl[1],0)
ha0=nz(hage[1],0)
hs=hrs and(h0~=1)
hy=hrb and(h0~=-1)
hbu=(h0==1)and(close>get_value(htl)+hb*0.15)and(close>open)and(corpo_atual>=hb*0.40)and(pavio_superior_pct<40)
hbd=(h0==-1)and(close<get_value(hfl)-hb*0.15)and(close<open)and(corpo_atual>=hb*0.40)and(pavio_inferior_pct<40)
if hs then htl:set(max(high,high[1]));hl:set(1);hage:set(0) elseif hy then hfl:set(min(low,low[1]));hl:set(-1);hage:set(0) elseif hbu or hbd then hl:set(0);hage:set(0) elseif h0~=0 then if ha0>=2 then hl:set(0);hage:set(0) else hage:set(ha0+1) end else hage:set(0) end
sinal_compra=sinal_compra and(not(((h0==1)and(not hbu))or hs))
sinal_venda=sinal_venda and(not(((h0==-1)and(not hbd))or hy))

-- REV apos H: confirmacao de rompimento da estrutura de 2 velas e corpo limpo.
-- Informativo. Nao se mistura aos candidatos de compra/venda.
local rev_idade_topo=make_series()
local rev_idade_fundo=make_series()
local rev_topo_usado=make_series()
local rev_fundo_usado=make_series()
local idade_topo=nz(rev_idade_topo[1],99)+1
local idade_fundo=nz(rev_idade_fundo[1],99)+1
local usado_topo=nz(rev_topo_usado[1],0)
local usado_fundo=nz(rev_fundo_usado[1],0)
if hs then idade_topo=0;usado_topo=0 end
if hy then idade_fundo=0;usado_fundo=0 end
local rev_corpo_ok=hb>0 and corpo_atual>=hb*.39 and range_atual>0 and corpo_atual/range_atual>=.54
local rev_venda=rev_on and idade_topo>=1 and idade_topo<=3 and usado_topo==0 and rev_corpo_ok and close<open and close<min(low[1],low[2])-hb*.04 and close<close[1] and pavio_inferior_pct<=41 and not pa_venda
local rev_compra=rev_on and idade_fundo>=1 and idade_fundo<=3 and usado_fundo==0 and rev_corpo_ok and close>open and close>max(high[1],high[2])+hb*.04 and close>close[1] and pavio_superior_pct<=41 and not pa_compra
if rev_venda and not rev_compra then usado_topo=1 elseif rev_compra and not rev_venda then usado_fundo=1 end
rev_idade_topo:set(idade_topo)
rev_idade_fundo:set(idade_fundo)
rev_topo_usado:set(usado_topo)
rev_fundo_usado:set(usado_fundo)

 if sinal_compra and not sinal_venda then b2_dir=1
 elseif sinal_venda and not sinal_compra then b2_dir=-1 end
