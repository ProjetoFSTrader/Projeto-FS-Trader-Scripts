    if close >= open then
        venda = false
    else
        compra = false
    end
end
plot_shape(compra, "Compra", shape_style.arrowup, shape_size.large, comprar_color, shape_location.belowbar, 0, "Compra", comprar_color)
plot_shape(venda, "Venda", shape_style.arrowdown, shape_size.large, vender_color, shape_location.abovebar, 0, "Venda", vender_color)
-- H: aviso de extremo/descanso, sem ordem automatica
h_compra = aviso_extremo_compra or s8_ponta_compra or s9_bloqueia_compra or f9_bloqueia_compra
h_venda  = aviso_extremo_venda  or s8_ponta_venda  or s9_bloqueia_venda  or f9_bloqueia_venda
plot_shape(h_compra, "H", shape_style.diamond, shape_size.large, "#FFFFFF", shape_location.abovebar, 0, "H", "#FFFFFF")
plot_shape(h_venda,  "H", shape_style.diamond, shape_size.large, "#FFFFFF", shape_location.belowbar, 0, "H", "#FFFFFF")
