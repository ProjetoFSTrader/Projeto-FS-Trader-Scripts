local ultimo_h=get_value(h_anterior[1],0)
local h_ref=get_value(h1anc[1],close[1])
local fpos=media_range_s8
local compra_forte=fpos>0 and close>max(high[1],h_ref)+fpos*.10
and close>open and corpo_atual>=fpos*.55
local venda_forte=fpos>0 and close<min(low[1],h_ref)-fpos*.10
and close<open and corpo_atual>=fpos*.55
local compra_livre=ultimo_h~=1 or compra_forte
local venda_livre=ultimo_h~=-1 or venda_forte
-- F9 EXPERIMENTAL: memoria de mudanca de estado apenas na correcao de extremos.
-- Estado confirmado na vela anterior; sem quarentena fixa nem filtro global.
local f9estado=make_series()
local f9dir_ant=get_value(f9estado[1],0)
local f9dir_ema=(close[1]>EMAA[1] and EMAA[1]>EMAA[2]) and 1 or ((close[1]<EMAA[1] and EMAA[1]<EMAA[2]) and -1 or 0)
local f9corr_top=f9dir_ant==1 and close[1]<open[1] and (high[1]>=highest(high[2],doch_time)-media_range_s8*.30)
local f9corr_fundo=f9dir_ant==-1 and close[1]>open[1] and (low[1]<=lowest(low[2],doch_time)+media_range_s8*.30)
-- Uma correcao no extremo invalida o rearme antigo ate retomada real ou REV.
local f9mem=make_series()
local f9bloq=get_value(f9mem[1],0)
if f9corr_top then f9bloq=1 elseif f9corr_fundo then f9bloq=-1 end
if rev_dir==-1 and f9bloq==1 then f9bloq=0 end
if rev_dir==1 and f9bloq==-1 then f9bloq=0 end
local f9retoma_c=close>high[1] and close>open and corpo_atual>media_range_s8*.40
local f9retoma_v=close<low[1] and close<open and corpo_atual>media_range_s8*.40
if f9bloq==1 and f9retoma_c then f9bloq=0 end
if f9bloq==-1 and f9retoma_v then f9bloq=0 end
f9mem:set(f9bloq)
f9estado:set(f9dir_ema)
if f9s1_d==0 and h1lado==0 and rev_dir==0 then
if not caixa and compra_livre and f9bloq~=1 and get_value(sinal_compra,false)then f9s1_d=1
elseif not caixa and venda_livre and f9bloq~=-1 and get_value(sinal_venda,false)then f9s1_d=-1 end
end
plot_shape(h1lado==1,"F9_H_TOPO",shape_style.arrowdown,shape_size.large,"#FF9800",shape_location.abovebar,0,"H","#FFFFFF")
plot_shape(h1lado==-1,"F9_H_FUNDO",shape_style.arrowup,shape_size.large,"#00E5FF",shape_location.belowbar,0,"H","#FFFFFF")
plot_shape(conditional(f9s1_d==1 and not pos_baleia),"F9_COMPRA",shape_style.arrowup,shape_size.large,call_color,shape_location.belowbar,0,"Compra","#fcfc04")
plot_shape(conditional(f9s1_d==-1 and not pos_baleia),"F9_VENDA",shape_style.arrowdown,shape_size.large,put_color,shape_location.abovebar,0,"Venda","#fcfc01")
-- Barras M15 normais: acompanham a ultima vela M15 fechada, sem memoria travada.
local m15_s=security(current_ticker_id,"15m")
input_group{"Barras M15",m15_cor_top=input{default="#FFFFFF",type=input.color},m15_cor_bot=input{default="#FFFFFF",type=input.color},m15_larg=input{default=2,type=input.line_width}}
if m15_s then
    plot(m15_s.high[1],"M15 Resistencia",m15_cor_top,m15_larg,0,style.levels,na_mode.continue)
    plot(m15_s.low[1],"M15 Suporte",m15_cor_bot,m15_larg,0,style.levels,na_mode.continue)
end
