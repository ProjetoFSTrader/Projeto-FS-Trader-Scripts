instrument { name = "FLUXO 2.0 NEON", icon="indicators:Bollinger_Bands", overlay = true }

-- =========================================================
-- FLUXO 2.0 LEVE
-- Ajuste simples para seguir fluxo e reduzir lateralizacao.
-- Usa apenas uma leitura curta inspirada no F9 (TA/TB + eficiencia),
-- sem H, L, R, memoria de estado ou arquitetura do FS.
-- =========================================================

input_group { "Visual",
    comprar_color     = input {default = "#00FF41", type = input.color},
    vender_color      = input {default = "#D50000", type = input.color},
    comprar_text_color = input {default = "#FFFFFF", type = input.color},
    vender_text_color  = input {default = "#00FFFF", type = input.color}
}

input_group { "Candles",
    candle_alta_color  = input {default = "#00FF41", type = input.color},
    candle_baixa_color = input {default = "#D50000", type = input.color},
    candle_neutro_color = input {default = "#9E9E9E", type = input.color}
}

input_group { "Fluxo",
    ema_micro_txt = input {default = "10", type = input.string},
    ema_fast_txt  = input {default = "3",  type = input.string},
    ema_slow_txt  = input {default = "13", type = input.string}
}

input_group { "Filtro de lateralizacao",
    eficiencia_min_txt       = input {default = "12", type = input.string},
    eficiencia_curta_min_txt = input {default = "8",  type = input.string},
    corpo_min_pct_txt        = input {default = "28", type = input.string}
}

-- [[ 1. MEDIAS / LEITURA DE FLUXO ]]
local ema_micro = ema(close, tonumber(ema_micro_txt))
local ema_fast  = ema(hlc3, tonumber(ema_fast_txt))
local ema_slow  = ema(hlc3, tonumber(ema_slow_txt))

-- Pedaco leve do conceito F9:
-- preco anda na direcao + fica do lado correto da EMA10 + EMA10 inclinada.
local TA = (close > close[1]) and (close > ema_micro) and (ema_micro > ema_micro[1])
local TB = (close < close[1]) and (close < ema_micro) and (ema_micro < ema_micro[1])

-- Confirmacao de fluxo. O rompimento da maxima/minima anterior permite
-- reconhecer inicio de perna sem esperar demais pelo cruzamento 3x13.
local fluxo_alta = TA and ((ema_fast >= ema_slow) or (close > high[1]))
local fluxo_baixa = TB and ((ema_fast <= ema_slow) or (close < low[1]))

-- [[ 2. CANDLES CORRIGIDOS ]]
-- Engolfo simples e coerente, usando a vela anterior contra a atual.
local eng_buy = (close[1] < open[1]) and (close > open) and
                (open <= close[1]) and (close >= open[1])

local eng_sell = (close[1] > open[1]) and (close < open) and
                 (open >= close[1]) and (close <= open[1])

-- Harami simples: corpo atual contido no corpo anterior.
local corpo_ant_top = max(open[1], close[1])
local corpo_ant_bot = min(open[1], close[1])
local corpo_at_top  = max(open, close)
local corpo_at_bot  = min(open, close)

local harami_buy = (close[1] < open[1]) and (close > open) and
                   (corpo_at_top <= corpo_ant_top) and (corpo_at_bot >= corpo_ant_bot)

local harami_sell = (close[1] > open[1]) and (close < open) and
                    (corpo_at_top <= corpo_ant_top) and (corpo_at_bot >= corpo_ant_bot)

local candle_buy = eng_buy or harami_buy
local candle_sell = eng_sell or harami_sell

-- [[ 3. LATERALIZACAO / FORCA ]]
local r0 = high-low
local r1 = high[1]-low[1]
local r2 = high[2]-low[2]
local r3 = high[3]-low[3]
local r4 = high[4]-low[4]
local r5 = high[5]-low[5]
local r6 = high[6]-low[6]
local r7 = high[7]-low[7]

local movimento_8 = r0+r1+r2+r3+r4+r5+r6+r7
local progresso_8 = abs(close-close[7])
local eficiencia_8 = (movimento_8 > 0) and (progresso_8/movimento_8*100) or 100
local lateralizado = eficiencia_8 < tonumber(eficiencia_min_txt)

local movimento_4 = r0+r1+r2+r3
local progresso_4 = abs(close-close[3])
local eficiencia_4 = (movimento_4 > 0) and (progresso_4/movimento_4*100) or 100
local lateralizado_curto = eficiencia_4 < tonumber(eficiencia_curta_min_txt)

local range_atual = high-low
local corpo_atual = abs(close-open)
local corpo_pct = (range_atual > 0) and (corpo_atual/range_atual*100) or 0
local corpo_ok = corpo_pct >= tonumber(corpo_min_pct_txt)

-- Evita comprar/vender vela praticamente sem deslocamento.
local media_range_8 = movimento_8 / 8
local deslocamento_ok = (media_range_8 > 0) and (corpo_atual >= media_range_8*0.18)

-- [[ 4. GATILHO ]]
-- Na tendencia limpa, permite continuidade.
-- Se o candle for um padrao conhecido, ele reforca a entrada, mas nao manda sozinho.
local continuidade_alta = close > high[1] and close > open
local continuidade_baixa = close < low[1] and close < open

local trigger_buy = fluxo_alta and (continuidade_alta or candle_buy)
local trigger_sell = fluxo_baixa and (continuidade_baixa or candle_sell)

-- [[ 5. SINAL FINAL ]]
local signal_long = trigger_buy and (not lateralizado) and (not lateralizado_curto) and corpo_ok and deslocamento_ok
local signal_short = trigger_sell and (not lateralizado) and (not lateralizado_curto) and corpo_ok and deslocamento_ok

-- Nunca deixa as duas direcoes aparecerem juntas na mesma vela.
local compra_final = signal_long and (not signal_short)
local venda_final  = signal_short and (not signal_long)

-- [[ 6. VISUAL NEON ]]
-- Candles: alta verde neon, baixa vermelho sangue e neutra cinza.
local candle_color
if close > open then
    candle_color = candle_alta_color
elseif close < open then
    candle_color = candle_baixa_color
else
    candle_color = candle_neutro_color
end

plot_candle(open, high, low, close, "FLUXO_NEON", candle_color)

plot_shape(compra_final, 'COMPRA', shape_style.arrowup, shape_size.large,
           comprar_color, shape_location.belowbar, 0, "COMPRA", comprar_text_color)

plot_shape(venda_final, 'VENDA', shape_style.arrowdown, shape_size.large,
           vender_color, shape_location.abovebar, 0, "VENDA", vender_text_color)
